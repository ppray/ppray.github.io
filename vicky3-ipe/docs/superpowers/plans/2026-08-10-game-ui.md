```javascript
/**
 * 《帝国的账本》海图渲染模块
 * 输出：renderMap(state, container, callbacks)
 */

import { BUILDINGS } from './data/buildings.js';
import { GOODS } from './data/goods.js';

const NATION_SHAPES = {
    GBR: {
        path: 'M178 58 Q190 45 200 52 Q205 65 195 75 Q182 78 178 68 Z',
        cx: 190, cy: 62, fill: '#7a8aa0', label: '🇬🇧 英国'
    },
    PRS: {
        path: 'M205 60 Q220 52 230 62 Q232 78 218 85 Q205 80 205 60 Z',
        cx: 217, cy: 66, fill: '#8a8a92', label: '🇩🇪 普鲁士'
    },
    QING: {
        path: 'M290 65 Q330 50 355 75 Q360 110 330 135 Q295 140 280 110 Q275 85 290 65 Z',
        cx: 318, cy: 95, fill: '#b06a4a', label: '🇨🇳 大清'
    },
    USA: {
        path: 'M45 60 Q75 48 100 58 Q105 85 85 105 Q55 110 42 90 Q38 72 45 60 Z',
        cx: 72, cy: 80, fill: '#6a8a72', label: '🇺🇸 美国'
    }
};

const CONTINENTS = [
    // 北美
    { d: 'M30 70 Q50 40 90 45 Q130 50 125 90 Q120 130 85 145 Q50 155 35 120 Q25 95 30 70 Z', fill: '#8a9a6a' },
    // 欧洲
    { d: 'M175 55 Q195 35 215 45 Q235 40 245 60 Q250 80 235 95 Q215 105 195 95 Q175 85 175 55 Z', fill: '#a89a6a' },
    // 亚洲
    { d: 'M255 50 Q300 35 340 55 Q375 75 365 115 Q350 155 305 160 Q265 160 255 120 Q245 85 255 50 Z', fill: '#b08a5a' }
];

const OCEAN_LINES = [
    'M0 60 Q100 55 200 60 T400 58',
    'M0 110 Q100 105 200 110 T400 108',
    'M0 170 Q100 165 200 170 T400 168',
    'M0 220 Q100 215 200 220 T400 218'
];

const ROUTES = [
    { from: 'QING', to: 'GBR', label: '生丝/茶' },
    { from: 'USA', to: 'GBR', label: '棉花' },
    { from: 'PRS', to: 'GBR', label: '钢铁/机械' }
];

const BUILDING_ICONS = {
    iron_mine: '⛏️',
    coal_mine: '⚒️',
    steel_mill: '🏭',
    textile_mill: '🧵',
    arms_factory: '🔫',
    tool_works: '⚙️',
    farm: '🚜',
    cotton_plantation: '🌿',
    shipping: '⛵'
};

export function renderMap(state, container, callbacks = {}) {
    container.innerHTML = '';
    const playerNation = state.playerNationKey;

    const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
    svg.setAttribute('viewBox', '0 0 400 250');
    svg.setAttribute('preserveAspectRatio', 'xMidYMid slice');

    // 海洋
    const ocean = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
    ocean.setAttribute('width', '400');
    ocean.setAttribute('height', '250');
    ocean.setAttribute('fill', '#c9b184');
    svg.appendChild(ocean);

    // 海图纬度线
    OCEAN_LINES.forEach(d => {
        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', d);
        path.setAttribute('stroke', '#a8905e');
        path.setAttribute('stroke-width', '0.7');
        path.setAttribute('fill', 'none');
        path.setAttribute('opacity', '0.5');
        svg.appendChild(path);
    });

    // 大陆轮廓（底层）
    CONTINENTS.forEach(c => {
        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', c.d);
        path.setAttribute('fill', c.fill);
        path.setAttribute('stroke', '#5a4a2a');
        path.setAttribute('stroke-width', '1');
        svg.appendChild(path);
    });

    // 国家高亮区
    Object.entries(NATION_SHAPES).forEach(([code, shape]) => {
        const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
        g.classList.add('map-nation-group');
        g.dataset.nation = code;

        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', shape.path);
        path.setAttribute('fill', shape.fill);
        path.setAttribute('stroke', code === playerNation ? '#e8c060' : '#3a2f1a');
        path.setAttribute('stroke-width', code === playerNation ? '2.5' : '1.5');
        path.classList.add('map-land');
        path.addEventListener('click', () => callbacks.onNationClick && callbacks.onNationClick(code));
        path.addEventListener('mouseenter', (e) => showTooltip(e, `${shape.label}<br>${state.nations[code].name}`));
        path.addEventListener('mouseleave', hideTooltip);
        g.appendChild(path);

        const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        text.setAttribute('x', shape.cx);
        text.setAttribute('y', shape.cy);
        text.classList.add('map-nation-label');
        text.textContent = code === 'GBR' ? '🇬🇧' : code === 'PRS' ? '🇩🇪' : code === 'QING' ? '🇨🇳' : '🇺🇸';
        g.appendChild(text);

        svg.appendChild(g);
    });

    // 贸易航线
    ROUTES.forEach(route => {
        const a = NATION_SHAPES[route.from];
        const b = NATION_SHAPES[route.to];
        const mx = (a.cx + b.cx) / 2;
        const my = Math.min(a.cy, b.cy) - 25;
        const d = `M${a.cx} ${a.cy} Q${mx} ${my} ${b.cx} ${b.cy}`;

        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', d);
        path.classList.add('map-route');
        svg.appendChild(path);
    });

    // 建筑图标
    Object.entries(state.nations).forEach(([code, nation]) => {
        const shape = NATION_SHAPES[code];
        let iconCount = 0;
        Object.entries(nation.buildings || {}).forEach(([bId, count]) => {
            if (count <= 0) return;
            const icon = BUILDING_ICONS[bId];
            if (!icon) return;
            // 在国土范围内做简单网格分布
            const x = shape.cx - 18 + (iconCount % 3) * 18;
            const y = shape.cy + 12 + Math.floor(iconCount / 3) * 18;
            iconCount++;

            const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            text.setAttribute('x', x);
            text.setAttribute('y', y);
            text.classList.add('map-building-icon');
            text.textContent = icon;
            text.addEventListener('click', (e) => {
                e.stopPropagation();
                callbacks.onBuildingClick && callbacks.onBuildingClick(code, bId);
            });
            text.addEventListener('mouseenter', (e) => showTooltip(e, `${BUILDINGS[bId].name} x${count}`));
            text.addEventListener('mouseleave', hideTooltip);
            svg.appendChild(text);

            if (count >= 10) {
                const badge = document.createElementNS('http://www.w3.org/2000/svg', 'text');
                badge.setAttribute('x', x + 7);
                badge.setAttribute('y', y + 8);
                badge.setAttribute('font-size', '6');
                badge.setAttribute('fill', '#1a1208');
                badge.setAttribute('font-weight', 'bold');
                badge.textContent = count;
                svg.appendChild(badge);
            }
        });
    });

    // 罗盘装饰
    const compass = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    compass.setAttribute('transform', 'translate(370, 30)');
    compass.setAttribute('opacity', '0.6');
    compass.innerHTML = `
        <circle r="12" fill="none" stroke="#5a4a2a" stroke-width="1"/>
        <path d="M0 -10 L3 0 L0 10 L-3 0 Z" fill="#5a4a2a"/>
        <text y="-15" text-anchor="middle" font-size="8" fill="#5a4a2a">N</text>
    `;
    svg.appendChild(compass);

    // 标题水印
    const watermark = document.createElementNS('http://www.w3.org/2000/svg', 'text');
    watermark.setAttribute('x', '200');
    watermark.setAttribute('y', '240');
    watermark.setAttribute('text-anchor', 'middle');
    watermark.setAttribute('font-size', '10');
    watermark.setAttribute('fill', '#5a4a2a');
    watermark.setAttribute('font-style', 'italic');
    watermark.setAttribute('font-family', 'Georgia, serif');
    watermark.textContent = '～ MARE LIBERUM · 1836 ～';
    svg.appendChild(watermark);

    container.appendChild(svg);
}

function showTooltip(evt, html) {
    let tip = document.getElementById('map-tooltip');
    if (!tip) {
        tip = document.createElement('div');
        tip.id = 'map-tooltip';
        tip.className = 'map-tooltip';
        document.body.appendChild(tip);
    }
    tip.innerHTML = html;
    tip.style.display = 'block';
    const rect = evt.target.getBoundingClientRect();
    tip.style.left = `${rect.left + window.scrollX + 12}px`;
    tip.style.top = `${rect.top + window.scrollY - 8}px`;
}

function hideTooltip() {
    const tip = document.getElementById('map-tooltip');
    if (tip) tip.style.display = 'none';
}
```
